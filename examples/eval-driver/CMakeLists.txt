find_package(OpenSSL REQUIRED COMPONENTS Crypto)
if(NOT TIMESTRAL_EVAL_RUNTIME_COMMIT MATCHES "^[0-9a-f]+$" OR NOT TIMESTRAL_EVAL_SOURCE_SHA256 MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "Explicit lowercase hex eval-driver identities required")
endif()
string(LENGTH "${TIMESTRAL_EVAL_RUNTIME_COMMIT}" _commit_len)
string(LENGTH "${TIMESTRAL_EVAL_SOURCE_SHA256}" _source_len)
if(NOT _commit_len EQUAL 40 OR NOT _source_len EQUAL 64)
  message(FATAL_ERROR "Eval-driver identities must contain 40/64 hex digits")
endif()
add_library(timestral-eval-codec STATIC protocol.cpp)
target_include_directories(timestral-eval-codec PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(timestral-eval-codec PUBLIC timestral OpenSSL::Crypto)
add_executable(timestral-eval-driver main.cpp)
target_link_libraries(timestral-eval-driver PRIVATE timestral-eval-codec)
target_compile_definitions(timestral-eval-driver PRIVATE TSQ_RUNTIME_COMMIT="${TIMESTRAL_EVAL_RUNTIME_COMMIT}" TSQ_SOURCE_SHA256="${TIMESTRAL_EVAL_SOURCE_SHA256}")
